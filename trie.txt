    public:
    Node* link[26];
    bool flag=false;

    bool containKey(char ch){
    return link[ch-'a']!=NULL;
    }
    void get(char ch,Node* node)
    {
        link[ch-'a']=node;
    }

    void setEnd()
    {
        flag=true;
    }

    bool isEnd()
    {
        return flag;
    }
};
